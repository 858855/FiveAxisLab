cmake_minimum_required(VERSION 3.24)
project(FiveAxisLab VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Compatibility for pinned upstream releases")
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
 add_compile_options(-fmax-errors=3)
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Concurrent Test)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(OpenCASCADE REQUIRED)
find_package(libzip CONFIG REQUIRED)
find_package(VTK REQUIRED COMPONENTS CommonCore CommonDataModel IOXML)
find_package(Boost REQUIRED COMPONENTS serialization filesystem system program_options)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(ENABLE_DOUBLE_PRECISION ON CACHE BOOL "" FORCE)
set(CCD_HIDE_ALL_SYMBOLS OFF CACHE BOOL "" FORCE)
set(CCD_FOUND TRUE)
add_subdirectory(vendor/libccd EXCLUDE_FROM_ALL)
set(CCD_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/libccd/src;${CMAKE_CURRENT_BINARY_DIR}/vendor/libccd/src")
set(CCD_LIBRARY ccd)
set(FCL_WITH_OCTOMAP OFF CACHE BOOL "" FORCE)
set(FCL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FCL_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/fcl EXCLUDE_FROM_ALL)
set(OMPL_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(OMPL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(OMPL_BUILD_PYBINDINGS OFF CACHE BOOL "" FORCE)
set(OMPL_BUILD_PYTESTS OFF CACHE BOOL "" FORCE)
set(OMPL_REGISTRATION OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/ompl EXCLUDE_FROM_ALL)
find_library(TETGEN_LIBRARY NAMES tet tetgen REQUIRED)
add_library(fiveaxis_core STATIC
 src/core/types.cpp src/core/geometry.cpp src/core/step_import.cpp
 src/core/stress.cpp src/core/slicing.cpp src/core/collision.cpp
 src/core/controllers.cpp src/core/project.cpp src/core/orca_adapter.cpp src/core/process.cpp src/core/motion_preview.cpp)
target_include_directories(fiveaxis_core PUBLIC src ${OpenCASCADE_INCLUDE_DIR} vendor/fcl/include
 "${CMAKE_CURRENT_BINARY_DIR}/vendor/fcl/include" vendor/ompl/src "${CMAKE_CURRENT_BINARY_DIR}/vendor/ompl/src")
target_link_libraries(fiveaxis_core PUBLIC Qt6::Core Eigen3::Eigen fcl ompl PRIVATE
 TKDESTEP TKXCAF TKLCAF TKCAF TKCDF TKBRep TKMesh TKPrim TKTopAlgo TKShHealing TKMath TKernel
 ${TETGEN_LIBRARY} libzip::zip VTK::CommonCore VTK::CommonDataModel VTK::IOXML)
target_compile_definitions(fiveaxis_core PRIVATE TETLIBRARY)
if(MINGW)
 # FCL 0.7 annotates inline template definitions as dllimport; MinGW rejects those definitions.
 # Unannotated calls are resolved by the same import library, while inline definitions remain local.
 target_compile_definitions(fiveaxis_core PUBLIC FCL_STATIC_DEFINE)
endif()
if(MSVC)
 target_compile_options(fiveaxis_core PRIVATE /W4)
else()
 target_compile_options(fiveaxis_core PRIVATE -Wall -Wextra -Wno-deprecated-declarations)
endif()
set(DESKTOP_SOURCES src/app/main_window.cpp src/app/process_dialog.cpp src/app/process_editor.cpp src/app/workspace.cpp src/app/step_dialog.cpp src/app/axis_preview.cpp src/app/viewport.cpp src/app/resources.qrc)
add_executable(FiveAxisLab WIN32 src/app/main.cpp ${DESKTOP_SOURCES})
target_link_libraries(FiveAxisLab PRIVATE fiveaxis_core Qt6::Widgets Qt6::OpenGLWidgets Qt6::Concurrent)
add_executable(fiveaxis_cli src/cli/main.cpp)
target_link_libraries(fiveaxis_cli PRIVATE fiveaxis_core)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE fiveaxis_core Qt6::Test)
add_test(NAME core COMMAND core_tests)
add_executable(desktop_tests tests/desktop_tests.cpp ${DESKTOP_SOURCES})
target_link_libraries(desktop_tests PRIVATE fiveaxis_core Qt6::Test Qt6::Widgets Qt6::OpenGLWidgets Qt6::Concurrent)
add_test(NAME desktop COMMAND desktop_tests)
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/tmp")
set_tests_properties(core PROPERTIES ENVIRONMENT "TMP=${CMAKE_BINARY_DIR}/tmp;TEMP=${CMAKE_BINARY_DIR}/tmp")
set_tests_properties(desktop PROPERTIES ENVIRONMENT "TMP=${CMAKE_BINARY_DIR}/tmp;TEMP=${CMAKE_BINARY_DIR}/tmp")
if(WIN32)
 get_filename_component(QT_RUNTIME_PREFIX "${Qt6_DIR}/../../.." ABSOLUTE)
 if(EXISTS "${QT_RUNTIME_PREFIX}/share/qt6/plugins")
  set_property(TEST desktop APPEND PROPERTY ENVIRONMENT "QT_PLUGIN_PATH=${QT_RUNTIME_PREFIX}/share/qt6/plugins")
 elseif(EXISTS "${QT_RUNTIME_PREFIX}/plugins")
  set_property(TEST desktop APPEND PROPERTY ENVIRONMENT "QT_PLUGIN_PATH=${QT_RUNTIME_PREFIX}/plugins")
 endif()
endif()
install(TARGETS FiveAxisLab fiveaxis_cli RUNTIME DESTINATION .)
